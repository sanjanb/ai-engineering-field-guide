def seclargest(n):
    lar = float("-inf")
    slar = float("-inf")
    for i in n:
        lar = max(lar, i)
    for i in n:
        if i >slar and i != lar:
            slar = i

# Method - Optimal

```python
def seclargest(n):
    lar = float("-inf")
    slar = float("-inf")
    for i in n:
        if i > lar:
            slar = lar
            lar = i
        elif i > slar and i != lar:
            slar = i

    return slar
print(seclargest([1,2,7, 4, 2, 4, 7, 0]))
```

### Time complexity = O(N)
### Spave complexity = O(1)
