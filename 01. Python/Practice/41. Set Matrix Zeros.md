        for j in range(0, c):
            if n[i][j] == 0:
                markinf(n, i, j)
    
    for i in range(0, r):

            if n[i][j] == float("inf"):
                n[i][j] = 0
    return n
                
def markinf(matrix, r, c):
    row = len(matrix)
    col =  len(matrix[0])

    for i in range(0, row):
        if matrix[i][r] != 0:
            matrix[i][r] = float("inf")
    for j in range(0, col):
        if matrix[c][j] != 0:
            matrix[c][j] = float("inf")

# Optimal Solution
```python 
def mat(n):
    r = len(n)
    c =  len(n[0])
    rtrack = [0 for _ in range(0, r)]
    ctrack = [0 for _ in range(0, c)]

    for i in range(0, r):
        for j in range(0, c):
            if n[i][j] == 0:
                rtrack[i] = -1
                ctrack[j] = -1

    for i in range(0, r):
        for j in range(0, c):
            if rtrack[i] == -1 or ctrack[j] == -1:
                n[i][j] = 0
    return n
    


test = [[1, 1, 1],[1, 0, 1], [1, 1, 1]]
print(mat(test))
```

### Time Complexity: O(2(m*n))
### Time Complexity: O(n+m)
