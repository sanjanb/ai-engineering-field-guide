# Method Bruteforce
```python
def sep(nums):
    temp = []
    n = len(nums)
    for i in range(0, n):
        if i != 0:
            temp.append(i)
    nz = len(temp)
    for i in range(0, nz):
        nums[i] = temp[i]

    for i in range(nz, n):
        nums[i] = 0
    return nums

print(sep([1, 2, 0, 0, 4, 0, 6]))
```
### Time Complexity: O(N)
### Space Complexity: O(1)


```python
def sep(nums):
    temp = []
    zero = []
    for i in nums:
        if i != 0:
            temp.append(i)
        else:
            zero.append(i)
    nums[:] = temp+zero
    return nums

print(sep([1, 2, 0, 0, 4, 0, 6]))
```
### Time Complexity: O(N)
### Space Complexity: O(1)

# Mwthod Optimal
```python
def sep(nums):
    i = 0
    for j in range(0, len(nums)):
        if nums[j] != 0:
            nums[i], nums[j] = nums[j], nums[i]
            i += 1 
    return nums

print(sep([1, 2, 0, 0, 4, 0, 6]))
```
### Time Complexity: O(N)
### Space Complexity: O(1)
